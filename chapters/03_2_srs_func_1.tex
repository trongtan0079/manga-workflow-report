\section{Yêu cầu chức năng (Functional Requirements)}
\label{sec:srs_functional_requirements}

\subsection{Tổng quan chức năng hệ thống}
\label{subsec:srs_functional_overview}

Phần này mô tả cấu trúc tổng quan các chức năng của hệ thống Manga Creation Workflow and Publishing Management System thông qua luồng đi giữa các màn hình giao diện (Screens Flow), đặc tả chi tiết của từng màn hình (Screen Descriptions) và ma trận phân quyền truy cập giữa các Actor (Screen Authorization).

\subsubsection{Các chức năng nền (Non-Screen Functions)}
Ngoài các chức năng tương tác qua màn hình giao diện, hệ thống còn thực hiện một số chức năng chạy ngầm tự động nhằm đảm bảo luồng công việc diễn ra thông suốt:
\begin{itemize}
    \item \textbf{Tự động gửi thông báo:} Hệ thống tự động kích hoạt gửi thông báo đến các Actor tương ứng khi có sự kiện thay đổi trạng thái (ví dụ: Mangaka giao Task mới $\rightarrow$ thông báo đến Assistant; Assistant nộp Submission $\rightarrow$ thông báo đến Mangaka; Chapter hoàn thành $\rightarrow$ thông báo đến Editor).
    \item \textbf{Tự động tính toán xếp hạng:} Sau khi Editorial Board nhập dữ liệu bình chọn của độc giả cho một kỳ phát hành, hệ thống tự động chạy ngầm thuật toán cộng dồn điểm, sắp xếp thứ hạng của các Series và kết xuất dữ liệu bảng xếp hạng mới.
    \item \textbf{Ghi nhật ký hoạt động hệ thống (System Audit Logs):} Hệ thống tự động ghi lại lịch sử thao tác của các tài khoản (đăng nhập, cập nhật dữ liệu, xóa dữ liệu) để phục vụ công tác giám sát hệ thống của Admin.
\end{itemize}

\subsection{Xác thực (Authentication)}
\label{subsec:srs_auth}

\subsubsection{UC-01: Đăng nhập hệ thống}
\label{subsubsec:uc01_login}
\begin{itemize}
    \item \textbf{Actor chính:} Tất cả các Actor (Admin, Mangaka, Assistant, Tantou Editor, Editorial Board).
    \item \textbf{Mô tả ngắn gọn:} Cho phép người dùng xác thực danh tính để đăng nhập vào hệ thống và bắt đầu phiên làm việc.
    \item \textbf{Tiền điều kiện:} Người dùng đã được cấp tài khoản và tài khoản đang ở trạng thái hoạt động (active).
    \item \textbf{Hậu điều kiện:} Người dùng đăng nhập thành công, phiên làm việc (session) được thiết lập và hệ thống chuyển hướng người dùng đến Dashboard phù hợp với vai trò.
    \item \textbf{Luồng sự kiện chính (Basic Flow):}
        \begin{enumerate}
            \item Người dùng truy cập vào trang đăng nhập của hệ thống (SCR-01).
            \item Hệ thống hiển thị form đăng nhập yêu cầu nhập Email hoặc Tên đăng nhập (Username) và Mật khẩu.
            \item Người dùng nhập chính xác Email hoặc Username, nhập Mật khẩu và nhấn nút "Đăng nhập".
            \item Hệ thống kiểm tra dữ liệu đầu vào. Nếu là Email (chứa ký tự '@'), hệ thống truy vấn tài khoản theo Email, ngược lại truy vấn theo Username.
            \item Hệ thống thực hiện băm mật khẩu và đối chiếu bằng hàm \texttt{password\_\allowbreak verify()} với cơ sở dữ liệu để xác thực tài khoản.
            \item Hệ thống ghi nhận đăng nhập thành công, thiết lập thông tin người dùng vào Session (gồm \texttt{user\_id}, \texttt{username}, \texttt{full\_name}, \texttt{role\_id}, \texttt{role\_name}) và tái tạo Session ID để chống tấn công Session Fixation.
            \item Hệ thống chuyển hướng người dùng đến Dashboard tương ứng với vai trò (SCR-02 đến SCR-06).
        \end{enumerate}
    \item \textbf{Luồng sự kiện ngoại lệ (Exception Flows):}
        \begin{itemize}
            \item \textbf{Sai thông tin hoặc Mật khẩu:} Hệ thống hiển thị thông báo lỗi "Tên đăng nhập/email hoặc mật khẩu không chính xác." và giữ người dùng ở lại màn hình đăng nhập.
            \item \textbf{Tài khoản bị khóa hoặc chưa kích hoạt:} Nếu tài khoản khớp thông tin nhưng trạng thái khác "active" (ví dụ: "inactive" hoặc "banned"), hệ thống hiển thị thông báo "Tài khoản của bạn đã bị khóa hoặc chưa được kích hoạt." và từ chối đăng nhập.
        \end{itemize}
\end{itemize}

\subsubsection{UC-02: Đăng xuất}
\label{subsubsec:uc02_logout}
\begin{itemize}
    \item \textbf{Actor chính:} Tất cả các Actor.
    \item \textbf{Mô tả ngắn gọn:} Cho phép người dùng hủy phiên làm việc hiện tại và thoát khỏi hệ thống một cách an toàn.
    \item \textbf{Tiền điều kiện:} Người dùng đang ở trạng thái đăng nhập trên hệ thống.
    \item \textbf{Hậu điều kiện:} Phiên làm việc bị xóa bỏ, người dùng không thể thực hiện các chức năng yêu cầu quyền truy cập và hệ thống đưa người dùng quay lại màn hình Đăng nhập (SCR-01).
    \item \textbf{Luồng sự kiện chính (Basic Flow):}
        \begin{enumerate}
            \item Người dùng nhấn vào biểu tượng cá nhân ở góc phải màn hình và chọn "Đăng xuất".
            \item Hệ thống hiển thị hộp thoại yêu cầu xác nhận việc đăng xuất.
            \item Người dùng chọn "Xác nhận".
            \item Hệ thống thực hiện thu hồi Token phiên làm việc, xóa toàn bộ session lưu trữ trên trình duyệt.
            \item Hệ thống chuyển hướng người dùng về màn hình đăng nhập (SCR-01).
        \end{enumerate}
    \item \textbf{Luồng sự kiện ngoại lệ (Exception Flows):} Không có.
\end{itemize}

\subsection{Quản lý người dùng (User Management)}
\label{subsec:srs_user_mgmt}

\subsubsection{UC-03: Xem danh sách người dùng}
\label{subsubsec:uc03_view_users}
\begin{itemize}
    \item \textbf{Actor chính:} Admin.
    \item \textbf{Mô tả ngắn gọn:} Cho phép Admin theo dõi, tìm kiếm và lọc danh sách toàn bộ các tài khoản có trong hệ thống.
    \item \textbf{Tiền điều kiện:} Admin đã đăng nhập thành công vào hệ thống.
    \item \textbf{Hậu điều kiện:} Hệ thống hiển thị danh sách người dùng đầy đủ và chính xác trên màn hình SCR-07.
    \item \textbf{Luồng sự kiện chính (Basic Flow):}
        \begin{enumerate}
            \item Admin nhấn chọn menu "Quản lý người dùng" trên thanh điều hướng.
            \item Hệ thống thực hiện truy vấn cơ sở dữ liệu lấy thông tin tất cả các tài khoản người dùng hiện có.
            \item Hệ thống kết xuất dữ liệu và hiển thị lên màn hình (SCR-07) dưới dạng bảng gồm các cột: STT, Họ và tên, Email, Vai trò, Trạng thái (Hoạt động/Bị khóa) và Ngày tạo.
            \item Admin có thể nhập từ khóa tìm kiếm (theo tên hoặc email) hoặc chọn bộ lọc (theo vai trò, trạng thái) để lọc danh sách.
            \item Hệ thống tự động lọc và hiển thị danh sách kết quả phù hợp theo bộ lọc của Admin.
        \end{enumerate}
    \item \textbf{Luồng sự kiện ngoại lệ (Exception Flows):}
        \begin{itemize}
            \item \textbf{Lỗi kết nối CSDL:} Hệ thống không thể tải dữ liệu, hiển thị thông báo lỗi "Kết nối đến máy chủ thất bại. Vui lòng tải lại trang".
        \end{itemize}
\end{itemize}

\subsubsection{UC-04: Tạo tài khoản}
\label{subsubsec:uc04_create_user}
\begin{itemize}
    \item \textbf{Actor chính:} Admin.
    \item \textbf{Mô tả ngắn gọn:} Cho phép Admin khởi tạo tài khoản mới cho các nhân sự tham gia vào hệ thống (Mangaka, Assistant, Editor...).
    \item \textbf{Tiền điều kiện:} Admin đã đăng nhập thành công.
    \item \textbf{Hậu điều kiện:} Tài khoản mới được thêm vào CSDL và hệ thống gửi thông báo thông tin đăng nhập đến email của người dùng mới.
    \item \textbf{Luồng sự kiện chính (Basic Flow):}
        \begin{enumerate}
            \item Tại màn hình Quản lý người dùng (SCR-07), Admin nhấn chọn nút "Tạo tài khoản".
            \item Hệ thống hiển thị một form tạo mới yêu cầu nhập: Họ và tên, Email, Mật khẩu ban đầu và chọn Vai trò (Admin, Mangaka, Assistant, Tantou Editor, Editorial Board).
            \item Admin điền đầy đủ các thông tin cần thiết, chọn vai trò phù hợp và nhấn nút "Lưu".
            \item Hệ thống kiểm tra các ràng buộc dữ liệu: Email phải đúng định dạng, mật khẩu tối thiểu 8 ký tự, các trường bắt buộc không được để trống và Email không được trùng lặp trong hệ thống.
            \item Hệ thống thực hiện mã hóa mật khẩu, ghi nhận thông tin tài khoản mới vào cơ sở dữ liệu với trạng thái mặc định là "Hoạt động".
            \item Hệ thống hiển thị thông báo "Tạo tài khoản thành công" và cập nhật lại bảng danh sách người dùng.
        \end{enumerate}
    \item \textbf{Luồng sự kiện ngoại lệ (Exception Flows):}
        \begin{itemize}
            \item \textbf{Email bị trùng lặp:} Hệ thống báo lỗi "Email này đã được sử dụng bởi một tài khoản khác trong hệ thống" và yêu cầu nhập lại email khác.
            \item \textbf{Dữ liệu đầu vào không hợp lệ:} Hệ thống hiển thị các dòng cảnh báo màu đỏ bên dưới các trường nhập liệu sai quy định và không cho phép lưu tài khoản.
        \end{itemize}
\end{itemize}

\subsubsection{UC-05: Cập nhật thông tin người dùng}
\label{subsubsec:uc05_update_user}
\begin{itemize}
    \item \textbf{Actor chính:} Admin.
    \item \textbf{Mô tả ngắn gọn:} Cho phép Admin thay đổi thông tin cá nhân hoặc vai trò của các người dùng trong hệ thống.
    \item \textbf{Tiền điều kiện:} Admin đã đăng nhập thành công.
    \item \textbf{Hậu điều kiện:} Các thông tin chỉnh sửa của người dùng được lưu trữ và cập nhật thành công vào cơ sở dữ liệu.
    \item \textbf{Luồng sự kiện chính (Basic Flow):}
        \begin{enumerate}
            \item Trên bảng danh sách ở màn hình SCR-07, Admin tìm kiếm tài khoản cần sửa và nhấn nút "Sửa" ở cột thao tác.
            \item Hệ thống hiển thị form chỉnh sửa thông tin điền sẵn dữ liệu hiện tại của người dùng.
            \item Admin thực hiện chỉnh sửa các thông tin cần thiết (Họ tên, Vai trò) và nhấn nút "Cập nhật".
            \item Hệ thống kiểm tra tính hợp lệ của dữ liệu mới nhập.
            \item Hệ thống cập nhật các thay đổi vào cơ sở dữ liệu và hiển thị thông báo "Cập nhật thông tin người dùng thành công".
        \end{enumerate}
    \item \textbf{Luồng sự kiện ngoại lệ (Exception Flows):}
        \begin{itemize}
            \item \textbf{Dữ liệu trống:} Admin xóa hết thông tin ở các trường bắt buộc và nhấn Cập nhật. Hệ thống hiển thị thông báo nhắc nhở "Thông tin không được để trống" và giữ nguyên form.
        \end{itemize}
\end{itemize}

\subsubsection{UC-06: Khóa / Mở khóa tài khoản}
\label{subsubsec:uc06_toggle_user}
\begin{itemize}
    \item \textbf{Actor chính:} Admin.
    \item \textbf{Mô tả ngắn gọn:} Cho phép Admin vô hiệu hóa (khóa) hoặc kích hoạt lại (mở khóa) quyền truy cập hệ thống của một tài khoản.
    \item \textbf{Tiền điều kiện:} Admin đã đăng nhập thành công.
    \item \textbf{Hậu điều kiện:} Trạng thái tài khoản được cập nhật trong CSDL; người dùng có tài khoản bị khóa sẽ không thể đăng nhập ở phiên sau.
    \item \textbf{Luồng sự kiện chính (Basic Flow):}
        \begin{enumerate}
            \item Admin tìm kiếm tài khoản cần thay đổi trạng thái hoạt động trên màn hình SCR-07.
            \item Admin nhấn vào nút "Khóa" (nếu tài khoản đang Hoạt động) hoặc nút "Mở khóa" (nếu tài khoản đang Bị khóa) ở cột thao tác.
            \item Hệ thống hiển thị một hộp thoại thông báo yêu cầu xác nhận hành động: "Bạn có chắc chắn muốn khóa/mở khóa tài khoản của [Tên người dùng]?".
            \item Admin chọn "Đồng ý".
            \item Hệ thống cập nhật trạng thái hoạt động của tài khoản trong CSDL và đổi nhãn hiển thị trạng thái trên màn hình thành "Bị khóa" hoặc "Hoạt động".
            \item Hệ thống hiển thị thông báo xác nhận thành công.
        \end{enumerate}
    \item \textbf{Luồng sự kiện ngoại lệ (Exception Flows):}
        \begin{itemize}
            \item \textbf{Tự khóa tài khoản của chính mình:} Admin cố tình chọn khóa tài khoản của chính mình đang dùng để đăng nhập. Hệ thống phát hiện và hiển thị thông báo lỗi "Không thể tự khóa tài khoản của chính mình" để ngăn chặn sai sót hành vi.
        \end{itemize}
\end{itemize}
