\section{Yêu cầu phi chức năng (Non-Functional Requirements)}
\label{sec:srs_nfr}

\subsection{Hiệu năng (Performance)}
\label{subsec:nfr_performance}
\begin{itemize}
    \item Hệ thống phải đảm bảo thời gian phản hồi phù hợp đối với các thao tác thông thường như đăng nhập, tra cứu dữ liệu, cập nhật thông tin và gửi bài.
    \item Hệ thống phải hỗ trợ nhiều người dùng truy cập đồng thời mà không ảnh hưởng đáng kể đến hiệu năng xử lý.
\end{itemize}

\subsection{Bảo mật (Security)}
\label{subsec:nfr_security}
\begin{itemize}
    \item Hệ thống phải yêu cầu người dùng đăng nhập trước khi truy cập các chức năng nghiệp vụ.
    \item Thông tin tài khoản và mật khẩu phải được lưu trữ an toàn trong cơ sở dữ liệu.
    \item Hệ thống phải áp dụng cơ chế phân quyền để đảm bảo mỗi người dùng chỉ được truy cập các chức năng phù hợp với vai trò được cấp.
\end{itemize}

\subsection{Khả năng mở rộng (Scalability)}
\label{subsec:nfr_scalability}
\begin{itemize}
    \item Hệ thống phải được thiết kế theo hướng dễ dàng mở rộng khi số lượng người dùng, Series Manga hoặc dữ liệu phát sinh tăng lên.
    \item Cấu trúc hệ thống cần hỗ trợ bổ sung các chức năng mới trong tương lai mà không làm ảnh hưởng đáng kể đến các chức năng hiện có.
\end{itemize}

\subsection{Tính sẵn sàng (Availability)}
\label{subsec:nfr_availability}
\begin{itemize}
    \item Hệ thống phải đảm bảo khả năng hoạt động ổn định trong quá trình sử dụng.
    \item Người dùng có thể truy cập và sử dụng hệ thống trong điều kiện kết nối mạng bình thường.
    \item Hệ thống cần hạn chế tối đa các lỗi gây gián đoạn quá trình làm việc.
\end{itemize}

\subsection{Khả năng sử dụng (Usability)}
\label{subsec:nfr_usability}
\begin{itemize}
    \item Giao diện hệ thống phải đơn giản, trực quan và dễ sử dụng đối với các nhóm người dùng khác nhau.
    \item Các chức năng cần được bố trí hợp lý nhằm giúp người dùng dễ dàng thao tác và tìm kiếm thông tin.
    \item Hệ thống cần cung cấp các thông báo và hướng dẫn cần thiết trong quá trình sử dụng.
\end{itemize}

\subsection{Khả năng bảo trì (Maintainability)}
\label{subsec:nfr_maintainability}
\begin{itemize}
    \item Hệ thống phải được xây dựng với cấu trúc rõ ràng, dễ quản lý và dễ nâng cấp.
    \item Mã nguồn cần được tổ chức hợp lý nhằm hỗ trợ việc sửa lỗi, bảo trì và phát triển thêm các chức năng mới trong tương lai.
    \item Cơ sở dữ liệu phải được thiết kế nhất quán nhằm đảm bảo thuận lợi cho việc quản lý và cập nhật dữ liệu.
\end{itemize}
